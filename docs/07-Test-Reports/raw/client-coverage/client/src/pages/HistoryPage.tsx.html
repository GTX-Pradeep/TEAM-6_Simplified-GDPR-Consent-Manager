import { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import { describeError } from "../api/auth";
import { getMyHistory, type HistoryItem } from "../api/consents";
import { FormAlert } from "../components/AuthCard";
 
/** Groups items by calendar day (local time), keeping newest-first order. */
export function groupByDay(items: HistoryItem[]): { day: string; items: HistoryItem[] }[] {
  const groups: { day: string; items: HistoryItem[] }[] = [];
  for (const item of items) {
    const day = new Date(item.recordedAt).toLocaleDateString(undefined, { dateStyle: "full" });
    const last = groups[groups.length - 1];
    if (last && last.day === day) last.items.push(item);
    else groups.push({ day, items: [item] });
  }
  return groups;
}
 
/**
 * FR-12: the user's complete consent history as a timeline, newest first.
 * Records are append-only, so this list can only grow; nothing is ever edited.
 */
export function HistoryPage() {
  const [items, setItems] = useState<HistoryItem[] | null>(null);
  const [error, setError] = useState<string | null>(null);
 
  useEffect(() => {
    getMyHistory()
      .then(setItems)
      .catch((err) => setError(describeError(err).message));
  }, []);
 
  return (
    <div className="mx-auto max-w-3xl px-4 py-10 sm:px-6">
      <Link to="/dashboard" className="text-sm font-medium text-brand-700 hover:underline">
        &larr; Back to my choices
      </Link>
      <p className="mt-6 text-sm font-semibold uppercase tracking-wider text-brand-600">Your consent history</p>
      <h1 className="mt-1 text-3xl font-bold text-brand-900">Every decision you made</h1>
      <p className="mt-2 text-slate-600">
        Each grant or withdrawal is stored permanently with the time and the policy version you saw. Entries are never
        changed or deleted.
      </p>
 
      {error && (
        <div className="mt-6">
          <FormAlert message={error} />
        </div>
      )}
      {!items && !error && <div className="card mt-6 h-48 animate-pulse bg-slate-100" aria-hidden="true" />}
      {items && items.length === 0 && (
        <div className="card mt-6 text-slate-600">
          No decisions yet. Grant or withdraw consent on{" "}
          <Link to="/dashboard" className="font-medium text-brand-700 hover:underline">
            your dashboard
          </Link>
          .
        </div>
      )}
 
      {items && items.length > 0 && (
        <div className="mt-8 space-y-8">
          {groupByDay(items).map((g) => (
            <section key={g.day} aria-label={g.day}>
              <h2 className="text-sm font-semibold text-slate-500">{g.day}</h2>
              <ol className="mt-3 space-y-3 border-l-2 border-slate-200 pl-6">
                {g.items.map((h) => {
                  const granted = h.action === "GRANTED";
                  return (
                    <li key={h.id} className="relative">
                      <span
                        aria-hidden="true"
                        className={`absolute -left-[31px] top-4 h-3 w-3 rounded-full ring-4 ring-slate-50 ${
                          granted ? "bg-emerald-500" : "bg-amber-500"
                        }`}
                      />
                      <div className="card p-4">
                        <div className="flex flex-wrap items-center justify-between gap-2">
                          <p className="font-medium text-slate-900">
                            {granted ? "Granted" : "Withdrew"} consent for{" "}
                            <span className="text-brand-800">{h.purpose.name}</span>
                          </p>
                          <time dateTime={h.recordedAt} className="text-xs text-slate-500">
                            {new Date(h.recordedAt).toLocaleTimeString(undefined, { timeStyle: "short" })}
                          </time>
                        </div>
                        <p className="mt-1 text-xs text-slate-500">
                          Under privacy policy v{h.policyVersion.versionNo} &middot; record #{h.id}
                        </p>
                      </div>
                    </li>
                  );
                })}
              </ol>
            </section>
          ))}
        </div>
      )}
    </div>
  );
}
